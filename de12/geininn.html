1"""芸人マッチング (Streamlit版)
2
3機能:
4  1. OpenAI API で好みに合う芸人をおすすめ
5  2. YouTube検索へのリンクを表示(APIは使いません)
6  3. Wikipedia API で芸人の概要(事務所・結成年など)を取得
7  4. お気に入りを favorites.json に保存し、次回以降のおすすめに反映
8  5. Wikipedia本文をもとに、プロフィール・お笑いのAI分析・賞レース戦績を表示
9
10環境変数:
11  OPENAI_API_KEY     (必須)
12  OPENAI_MODEL       (任意。既定: gpt-6-luna)
13"""

26MODEL = os.environ.get("OPENAI_MODEL", "gpt-6-luna")
27FAV_FILE = Path("favorites.json")
28HEADERS = {"User-Agent": "GeinoMatching/0.1 (learning project)"}

30SYSTEM_PROMPT = """日本のお笑いに詳しいアドバイザーです。
31ユーザーの好みと、お気に入り済みの芸人の傾向をふまえ、おすすめの芸人(コンビ・ピン含む)を5組選んでください。
32お気に入り済みの芸人は除外してください。実在しない芸人は絶対に挙げないでください。
33出力はJSON配列のみ。前置きやコードブロックは不要です。各要素は次の形式:
34{"name": "芸人名", "style": "芸風", "reason": "おすすめ理由(好みと結びつけて2文以内)"}"""

37# お気に入りの保存のためにリストに自分で名前を追加できるようにしたが、実際のおすすめはOpenAI APIで取得するため、ここではお気に入りの保存・削除と読み込み・書き込みのみを行う。
38def load_favs() -> list[str]:
39    if FAV_FILE.exists():
40        try:
41            return json.loads(FAV_FILE.read_text(encoding="utf-8"))
42        except json.JSONDecodeError:
43            return []
44    return []
45
46
47def save_favs(favs: list[str]) -> None:
48    FAV_FILE.write_text(json.dumps(favs, ensure_ascii=False, indent=2), encoding="utf-8")
49
50
51def add_fav(name: str) -> None:
52    if name and name not in st.session_state.favs:
53        st.session_state.favs.append(name)
54        save_favs(st.session_state.favs)
55
56
57def remove_fav(name: str) -> None:
58    st.session_state.favs = [f for f in st.session_state.favs if f != name]
59    save_favs(st.session_state.favs)

62#  外部APIを使って検索する機能を追加
63@st.cache_data(show_spinner=False, ttl=3600)
64def recommend(preference: str, favs: tuple[str, ...]) -> list[dict]:
65    client = OpenAI()  # 環境変数 OPENAI_API_KEY を自動で読み込む
66    content = f"好み: {preference}\nお気に入り済み: {'、'.join(favs) or 'なし'}"
67    resp = client.chat.completions.create(
68        model=MODEL,
69        messages=[
70            {"role": "system", "content": SYSTEM_PROMPT},
71            {"role": "user", "content": content},
72        ],
73    )
74    text = resp.choices[0].message.content or ""
75    match = re.search(r"\[.*\]", text, re.S)
76    return json.loads(match.group(0)) if match else []

79@st.cache_data(show_spinner=False, ttl=86400)
80def wiki_summary(name: str) -> dict | None:
81    try:
82        r = requests.get(
83            "https://ja.wikipedia.org/w/api.php",
84            params={"action": "query", "list": "search", "srsearch": f"{name} お笑い",
85                    "srlimit": 1, "format": "json"},
86            headers=HEADERS, timeout=10,
87        )
88        r.raise_for_status()
89        hits = r.json()["query"]["search"]
90        if not hits:
91            return None
92        s = requests.get(
93            "https://ja.wikipedia.org/api/rest_v1/page/summary/" + quote(hits[0]["title"]),
94            headers=HEADERS, timeout=10,
95        )
96        s.raise_for_status()
97        d = s.json()
98        return {"title": d["title"], "extract": d.get("extract", ""),
99                "url": d["content_urls"]["desktop"]["page"]}
100    except (requests.RequestException, KeyError, ValueError):
101        return None

104ANALYZE_PROMPT = """あなたはお笑いの分析者です。与えられたWikipediaの記事本文だけを根拠に、芸人「__NAME__」について次のJSONのみを出力してください。前置きやコードブロックは不要です。
105記事に書かれていない事実は推測せず「記載なし」としてください。
106{
107  "profile": {"所属事務所": "", "結成・デビュー": "", "メンバー": "", "出身・経歴": "", "主な活動": ""},
108  "style_analysis": "どんなお笑いか(芸風・ネタの特徴・魅力)を3〜4文で。記事の内容に基づくこと",
109  "awards": [{"contest": "賞レース名", "year": "年", "result": "優勝・準優勝・決勝進出・準決勝進出 など"}]
110}
111awardsはM-1グランプリ、キングオブコント、R-1グランプリ、THE W、ABCお笑いグランプリ、NHK新人お笑い大賞など
112賞レースの戦績で、記事に記載があるものだけを新しい年順に並べてください。なければ空配列にしてください。"""
113
114
115@st.cache_data(show_spinner=False, ttl=86400)
116def wiki_fulltext(title: str) -> str:
117    try:
118        r = requests.get(
119            "https://ja.wikipedia.org/w/api.php",
120            params={"action": "query", "prop": "extracts", "explaintext": 1, "redirects": 1,
121                    "titles": title, "format": "json"},
122            headers=HEADERS, timeout=10,
123        )
124        r.raise_for_status()
125        page = next(iter(r.json()["query"]["pages"].values()))
126        return page.get("extract", "")[:25000]
127    except (requests.RequestException, KeyError, ValueError, StopIteration):
128        return ""
129
130
131@st.cache_data(show_spinner=False, ttl=86400)
132def analyze_comedian(name: str, title: str) -> dict:
133    text = wiki_fulltext(title)
134    if not text:
135        return {}
136    client = OpenAI()
137    resp = client.chat.completions.create(
138        model=MODEL,
139        messages=[
140            {"role": "system", "content": ANALYZE_PROMPT.replace("__NAME__", name)},
141            {"role": "user", "content": f"【Wikipedia記事: {title}】\n{text}"},
142        ],
143    )
144    match = re.search(r"\{.*\}", resp.choices[0].message.content or "", re.S)
145    return json.loads(match.group(0)) if match else {}

148def show_detail(name: str) -> None:
149    wiki = wiki_summary(name)
150    if not wiki:
151        st.info("Wikipediaに該当する記事が見つかりませんでした。")
152        return
153    with st.spinner("記事を読んで分析中です…"):
154        try:
155            info = analyze_comedian(name, wiki["title"])
156        except (OpenAIError, json.JSONDecodeError) as e:
157            st.error(f"分析に失敗しました: {e}")
158            return
159    if not info:
160        st.info("記事の本文を取得できませんでした。")
161        return
162    tab1, tab2, tab3 = st.tabs(["👤 プロフィール", "🎭 お笑い分析", "🏆 賞レース戦績"])
163    with tab1:
164        if wiki["extract"]:
165            st.write(wiki["extract"])
166        for k, v in (info.get("profile") or {}).items():
167            st.markdown(f"- **{k}**：{v or '記載なし'}")
168    with tab2:
169        st.write(info.get("style_analysis") or "分析できませんでした。")
170    with tab3:
171        awards = info.get("awards") or []
172        if awards:
173            st.table([{"賞レース": a.get("contest", ""), "年": a.get("year", ""),
174                       "結果": a.get("result", "")} for a in awards])
175        else:
176            st.write("記事に賞レースの戦績の記載は見つかりませんでした。")
177    st.caption(f"出典: [Wikipedia「{wiki['title']}」]({wiki['url']})をもとにAIが抽出・要約しています。"
178               "誤りや、同名の別人の記事を参照している可能性があるため、必ず出典でご確認ください。")
